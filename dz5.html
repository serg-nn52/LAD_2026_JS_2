function limitStr(str, n, symb = '...') {

    return str.slice(0, n) + symb;

  return str;
}

№2.6 — другая задача: нужно перевести строку из PascalCase/camelCase: initSnake("HelloMyWorld") → "hello_my_world". В такой строке нет пробелов, поэтому split(" ") ничего не делит и получается "hellomyworld". Перед каждой заглавной буквой (кроме первой) нужно ставить _ — перебором строки, как ты сделала в №2.3:

function initSnake(str) {
  let result = '';
  for (let i = 0; i < str.length; i++) {
    const isUpper = str[i] !== str[i].toLowerCase();
    if (isUpper && i > 0) {
      result += '_';
    }
    result += str[i].toLowerCase();
  }
  return result;
}
